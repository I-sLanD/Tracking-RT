cmake_minimum_required(VERSION 3.20)
project(jetson_perception_cpp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(ORT_INCLUDE_DIR "${CMAKE_SOURCE_DIR}/third_party/onnxruntime-directml/build/native/include" CACHE PATH "ONNX Runtime headers")
add_executable(yolo26_verify src/main.cpp)
set(EIGEN_INCLUDE_DIR "${CMAKE_SOURCE_DIR}/third_party/eigen-3.4.0" CACHE PATH "Eigen headers")
add_library(deepsort STATIC src/deepsort.cpp)
target_include_directories(deepsort PUBLIC "${EIGEN_INCLUDE_DIR}" "${CMAKE_SOURCE_DIR}/src")
add_executable(yolo26_track src/track_main.cpp)
target_link_libraries(yolo26_track PRIVATE deepsort)
include(CTest)
if(BUILD_TESTING)
  add_executable(tracker_tests tests/tracker_tests.cpp)
  target_link_libraries(tracker_tests PRIVATE deepsort)
  add_test(NAME deepsort_algorithm COMMAND tracker_tests)
endif()
foreach(target yolo26_verify yolo26_track)
  target_include_directories(${target} PRIVATE "${ORT_INCLUDE_DIR}" "${CMAKE_SOURCE_DIR}/third_party/include")
  if(MINGW)
    target_link_options(${target} PRIVATE -static -municode)
  elseif(UNIX)
    target_link_libraries(${target} PRIVATE ${CMAKE_DL_LIBS})
  endif()
endforeach()
foreach(target yolo26_verify yolo26_track deepsort tracker_tests)
  if(TARGET ${target})
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /utf-8)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
  endif()
endforeach()
if(MINGW AND TARGET tracker_tests)
  target_link_options(tracker_tests PRIVATE -static)
endif()
